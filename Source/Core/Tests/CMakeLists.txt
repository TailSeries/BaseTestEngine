# Standalone validation: cmake -S Source/Core/Tests -B out/build/core-lockfree-tests
cmake_minimum_required(VERSION 3.20)
project(CoreLockfreeTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# Keep the DLL beside test executables, including after a DLL-only rebuild.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
enable_testing()
find_package(Threads REQUIRED)
if(WIN32)
    add_compile_definitions(NOMINMAX SYSTEM_WIN PLATFORM_WIN)
    add_compile_options($<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
    # Build the actual Core DLL for end-to-end TaskGraph tests.
    add_subdirectory(.. CoreRuntime)
else()
    add_subdirectory(../ThirdParty/BoostLockfree BoostLockfree)
endif()
add_executable(CoreLockfreeTests LockfreeTests.cpp)
target_compile_features(CoreLockfreeTests PRIVATE cxx_std_20)
target_link_libraries(CoreLockfreeTests PRIVATE CoreThirdParty::BoostLockfree Threads::Threads)
add_test(NAME CoreLockfreeTests COMMAND CoreLockfreeTests)
set_tests_properties(CoreLockfreeTests PROPERTIES TIMEOUT 45)

if(WIN32)
    add_executable(CoreLogTests LogTests.cpp LogTestCategory.cpp)
    target_link_libraries(CoreLogTests PRIVATE Core Threads::Threads)
    add_test(NAME CoreLogTests COMMAND CoreLogTests)
    set_tests_properties(CoreLogTests PROPERTIES TIMEOUT 30)

    add_executable(CoreStallingQueueTests StallingTaskQueueTests.cpp)
    target_compile_features(CoreStallingQueueTests PRIVATE cxx_std_20)
    target_compile_definitions(CoreStallingQueueTests PRIVATE NOMINMAX SYSTEM_WIN PLATFORM_WIN)
    target_compile_options(CoreStallingQueueTests PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
    target_include_directories(CoreStallingQueueTests PRIVATE .. ../Base ../Common)
    target_link_libraries(CoreStallingQueueTests PRIVATE CoreThirdParty::BoostLockfree Threads::Threads)
    add_test(NAME CoreStallingQueueTests COMMAND CoreStallingQueueTests)
    set_tests_properties(CoreStallingQueueTests PROPERTIES TIMEOUT 45)

    add_executable(CoreTaskGraphTests TaskGraphTests.cpp)
    target_link_libraries(CoreTaskGraphTests PRIVATE Core Threads::Threads)
    add_test(NAME CoreTaskGraphTests COMMAND CoreTaskGraphTests)
    set_tests_properties(CoreTaskGraphTests PROPERTIES TIMEOUT 60)
endif()
